cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}" CACHE PATH "Pico SDK directory")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(bldc_motorcontrol C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.c" MAIN_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/web_policy.h" POLICY_HASH)
string(SHA256 BUILD_HASH "${MAIN_HASH}:${POLICY_HASH}:${PICO_SDK_VERSION_STRING}")
string(SUBSTRING "${BUILD_HASH}" 0 16 FIRMWARE_BUILD_ID)
configure_file(build_id.h.in build_id.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS main.c web_policy.h)
add_executable(motorcontrol_bldc main.c ../vendor/cJSON.c)
target_include_directories(motorcontrol_bldc PRIVATE ../vendor ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(motorcontrol_bldc PRIVATE CJSON_NESTING_LIMIT=4 PICO_STDIO_USB_STDOUT_TIMEOUT_US=10000 PICO_FLASH_SIZE_BYTES=2097152)
target_compile_options(motorcontrol_bldc PRIVATE -Wall -Wextra -Werror)
target_link_libraries(motorcontrol_bldc pico_stdlib pico_unique_id hardware_adc hardware_i2c hardware_watchdog m)
pico_enable_stdio_usb(motorcontrol_bldc 1)
pico_enable_stdio_uart(motorcontrol_bldc 0)
pico_add_extra_outputs(motorcontrol_bldc)
